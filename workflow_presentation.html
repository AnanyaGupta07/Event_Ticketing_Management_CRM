    %% 3. APPROVAL ENGINE
    subgraph APPROVAL["3. DYNAMIC MULTI-TIER CATEGORY APPROVAL ENGINE"]
        AP1[EventApprovalService.cls
Evaluates Category Ceilings via Approval_Matrix__mdt]
        O8 --> AP1
        
        AP1 --> D_AUTO{Proposed Budget <=
Category Auto-Limit?}
        
        D_AUTO -- "YES (Tier 1: Auto-Approved)" --> AP_OK[Approval_Status__c = Approved]
        
        D_AUTO -- "NO (Requires Review)" --> AP_SUB[Submit to Salesforce Approval Process
Status = Pending Approval
Record Locked]
        
        AP_SUB --> D_ESC{Budget > Manager Limit?}
        D_ESC -- "YES (Tier 3)" --> AP_FIN[Flag Requires_Finance_Signoff = TRUE
Escalate to Executive Signoff]
        D_ESC -- "NO (Tier 2)" --> AP_MGR[Route to Event Manager Queue/User]
        
        AP_FIN --> D_DEC{Approver Decision}
        AP_MGR --> D_DEC
        
        D_DEC -- "APPROVE" --> AP_OK
        D_DEC -- "REJECT" --> AP_REJ[Approval_Status__c = Rejected
Approver Enters Rejection_Reason__c]
        
        AP_REJ --> D_RESUB{Organizer Action}
        D_RESUB -- "Modify Budget & Resubmit" --> AP1
        D_RESUB -- "Cancel Event" --> AP_CANCEL([Approval_Status__c = Cancelled])
    end

        AT5 --> AT6[Assign Shared Booking_Group_Id__c

    end

    %% 6. UPI PAYMENT
    subgraph PAYMENT["6. SIMULATED UPI QR PAYMENT & SETTLEMENT"]
        AT6 --> PM1[Open LWC paymentQrVerification

        PM1 --> PM2[Generate Dynamic UPI Intent:
upi://pay?pa=events@upi&pn=EventHub&am=Total&tr=Group_Id]
        PM2 --> PM3[Render QR Code inside 170px SVG Countdown Ring
10-Second Auto-Verification Timer]
        PM3 --> PM4{Countdown Finishes OR
Attendee clicks I Have Paid}
        PM4 --> PM5[Call EventBookingController.confirmPayment]
        PM5 --> D_IDEMP{Idempotency Guard:
Payment already exists?}
        D_IDEMP -- "YES" --> PM_SKIP([Prevent Duplicate Payment])
        D_IDEMP -- "NO" --> PM6[Insert Payment__c Record:
Status = Successful, Method = UPI]
        PM6 --> PM7[Atomically Flip All Registrations in Group:
Status: Pending -> Confirmed]
        PM7 --> PM8[Generate Official Ticket__c Entry Passes:

    end
